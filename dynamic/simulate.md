## 🐍 `adversary.py` (Updated with Learning & Logging)

```python
import random
import json
import os

class Adversary:
    def __init__(self, name="Hydra", adv_type="adaptive", memory_file="memory.json"):
        self.name = name
        self.type = adv_type
        self.history = []
        self.memory_file = memory_file
        self.load_memory()

    def respond(self, user_choice):
        self.history.append(user_choice)

        if self.type == "adaptive":
            return self.adapt()
        elif self.type == "predictive":
            return self.predict_next()
        elif self.type == "chaotic":
            return self.random_choice()
        elif self.type == "mimic":
            return self.mirror()

        return "Exploit"

    def adapt(self):
        # If user is succeeding a lot with a strategy, counter it
        conserve_wins = sum(1 for x in self.memory if x["user"] == "Conserve" and x["outcome"] == "success")
        exploit_wins = sum(1 for x in self.memory if x["user"] == "Exploit" and x["outcome"] == "success")
        return "Exploit" if conserve_wins > exploit_wins else "Conserve"

    def predict_next(self):
        return "Exploit" if self.history.count("Conserve") > self.history.count("Exploit") else "Conserve"

    def mirror(self):
        return self.history[-1] if self.history else "Conserve"

    def random_choice(self):
        return random.choice(["Conserve", "Exploit"])

    def load_memory(self):
        if os.path.exists(self.memory_file):
            with open(self.memory_file, 'r') as f:
                self.memory = json.load(f)
        else:
            self.memory = []

    def log_interaction(self, user_choice, adv_choice, outcome):
        self.memory.append({
            "user": user_choice,
            "adversary": adv_choice,
            "outcome": outcome
        })
        with open(self.memory_file, 'w') as f:
            json.dump(self.memory, f, indent=2)
```

## 🐍 `survival.py`

Estimates Kaplan-Meier from memory.json

```python
import json
import pandas as pd
from lifelines import KaplanMeierFitter

def load_memory(memory_file='memory.json'):
    with open(memory_file, 'r') as f:
        return json.load(f)

def build_km_curve():
    memory = load_memory()
    times = []
    events = []

    # Simulated time/event data: 1 unit per round
    for i, entry in enumerate(memory):
        times.append(i + 1)
        events.append(1 if entry["outcome"] == "loss" else 0)

    df = pd.DataFrame({'time': times, 'event': events})
    if df.empty:
        return [], []

    kmf = KaplanMeierFitter()
    kmf.fit(df['time'], event_observed=df['event'])

    # Return x and y for Plotly
    return kmf.survival_function_.index.tolist(), kmf.survival_function_['KM_estimate'].tolist()
```

from flask import Flask, render_template, request, redirect
import yaml, json, random
from datetime import datetime
from lifelines import KaplanMeierFitter
import pandas as pd

app = Flask(__name__)

# ----- Data Loaders -----

def load_patient():
    return {
        "id": "P001",
        "age": 55,
        "race": "Black",
        "has_living_donor": True
    }

def load_memory(file="memory.json"):
    try:
        with open(file) as f:
            return json.load(f)
    except:
        return []

def save_memory(memory, file="memory.json"):
    with open(file, "w") as f:
        json.dump(memory, f, indent=2)

def log_decision(user, adv, outcome):
    memory = load_memory()
    memory.append({
        "timestamp": datetime.now().isoformat(),
        "user": user,
        "adv": adv,
        "outcome": outcome
    })
    save_memory(memory)

def load_logic_file(path="logic.yml"):
    with open(path) as f:
        return f.read()

def save_logic_file(content, path="logic.yml"):
    with open(path, "w") as f:
        f.write(content)

# ----- KM Curves -----

def build_km_curve(memory):
    times, events = [], []
    for i, m in enumerate(memory):
        times.append(i + 1)
        events.append(1 if m["outcome"].lower() == "loss" else 0)
    if not times:
        return [], []
    df = pd.DataFrame({"time": times, "event": events})
    kmf = KaplanMeierFitter()
    kmf.fit(df["time"], event_observed=df["event"])
    return kmf.survival_function_.index.tolist(), kmf.survival_function_["KM_estimate"].tolist()

# ----- ROUTE: /alive -----

@app.route("/alive")
def alive():
    patient = load_patient()
    memory = load_memory()
    times, survival = build_km_curve(memory)

    # fake counterfactual: 30% higher event rate
    cf_events = [{"time": m["timestamp"], "event": 1 if m["outcome"] == "loss" or random.random() < 0.3 else 0} for m in memory]
    cf_times, cf_survival = build_km_curve(cf_events)

    return render_template("alive.jinja",
        patient=patient,
        memory=memory,
        times=times,
        survival=survival,
        cf_times=cf_times,
        cf_survival=cf_survival
    )

# ----- ROUTE: /edit -----

@app.route("/edit", methods=["GET", "POST"])

    if request.method == "POST":
        save_logic_file(request.form["logic"])
        return redirect("/edit")
    logic = load_logic_file()
    return render_template("edit.jinja", logic=logic)

# ----- ROUTE: /play -----

@app.route("/play")
def play():
    groups = {"With Donor": [], "No Donor": []}
    for _ in range(1000):
        donor = random.choice([True, False])
        survival = []
        for i in range(10):
            # with donor survives longer
            death_chance = 0.1 if donor else 0.3
            survival.append(0 if random.random() < death_chance else 1)
        for i, e in enumerate(survival):
            groups["With Donor" if donor else "No Donor"].append({"time": i+1, "event": 1 - e})

    km_data = {}
    for label, group in groups.items():
        if not group:
            km_data[label] = {"times": [], "survival": []}
            continue
        df = pd.DataFrame(group)
        kmf = KaplanMeierFitter()
        kmf.fit(df["time"], df["event"])
        km_data[label] = {
            "times": kmf.survival_function_.index.tolist(),
            "survival": kmf.survival_function_["KM_estimate"].tolist()
        }

    return render_template("play.jinja", km_data=km_data)

from lifelines import CoxPHFitter
import pandas as pd

def run_cox_model(records):
    """
    Input: list of dicts with keys 'time', 'event', plus covariates (e.g. 'age', 'has_donor')
    Output: Cox model summary + survival predictions
    """
    df = pd.DataFrame(records)
    if df.empty or df["event"].sum() == 0:
        return None, None

    cph = CoxPHFitter()
    cph.fit(df, duration_col="time", event_col="event")

    return cph.summary, cph.predict_survival_function(df)
```

from lifelines import CoxTimeVaryingFitter
import pandas as pd

def run_tv_cox_model(records):
    df = pd.DataFrame(records)
    ctv = CoxTimeVaryingFitter()
    ctv.fit(df, id_col="id", start_col="start", stop_col="stop", event_col="event")
    return ctv.summary
```
